================================================================================
                    LARAVEL BACKEND API ARCHITECTURE
================================================================================

┌─────────────────────────────────────────────────────────────────────────────┐
│                         HTTP REQUEST FLOW                                   │
├─────────────────────────────────────────────────────────────────────────────┤
│                                                                              │
│  GET /api/products?search=laptop&categoryId=1&warehouseId=1&perPage=50     │
│                              ↓                                              │
│  ┌───────────────────────────────────────────────────────────────────────┐  │
│  │ Route Handler (routes/api.php)                                        │  │
│  │ - Matches: apiResource('products', ProductController::class)         │  │
│  │ - Routes to: ProductController@index()                               │  │
│  └──────────────────────────┬──────────────────────────────────────────┘  │
│                              ↓                                              │
│  ┌───────────────────────────────────────────────────────────────────────┐  │
│  │ ProductController::index(Request $request)                            │  │
│  │ ┌─────────────────────────────────────────────────────────────────┐  │  │
│  │ │ 1. Initialize Query:                                            │  │  │
│  │ │    $query = Product::with(['category', 'warehouse'])           │  │  │
│  │ │                                                                  │  │  │
│  │ │ 2. Apply Search Scope (via Model):                             │  │  │
│  │ │    ->search($request->search)                                   │  │  │
│  │ │    ↓ Executes scopeSearch() in Product model                   │  │  │
│  │ │                                                                  │  │  │
│  │ │ 3. Apply Category Filter:                                       │  │  │
│  │ │    ->where('category_id', $request->categoryId)                │  │  │
│  │ │                                                                  │  │  │
│  │ │ 4. Apply Warehouse Filter:                                      │  │  │
│  │ │    ->where('warehouse_id', $request->warehouseId)              │  │  │
│  │ │                                                                  │  │  │
│  │ │ 5. Apply Sorting:                                               │  │  │
│  │ │    ->latest()  [created_at DESC]                                │  │  │
│  │ │                                                                  │  │  │
│  │ │ 6. Apply Pagination:                                            │  │  │
│  │ │    ->paginate($request->perPage ?? 50)                          │  │  │
│  │ │                                                                  │  │  │
│  │ │ 7. Transform Response:                                          │  │  │
│  │ │    return ProductResource::collection($products)                │  │  │
│  │ └─────────────────────────────────────────────────────────────────┘  │  │
│  └───────────────────────────────────────────────────────────────────────┘  │
│                              ↓                                              │
│  ┌───────────────────────────────────────────────────────────────────────┐  │
│  │ Product Model (app/Models/Product.php)                               │  │
│  │ ┌─────────────────────────────────────────────────────────────────┐  │  │
│  │ │ scopeSearch($query, $search)                                    │  │  │
│  │ │   WHERE name LIKE '%laptop%' OR code LIKE '%laptop%'            │  │  │
│  │ │                                                                  │  │  │
│  │ │ Relationships:                                                  │  │  │
│  │ │   - belongsTo(Category)                                         │  │  │
│  │ │   - belongsTo(Warehouse)                                        │  │  │
│  │ │   - hasMany(TransactionDetail)                                  │  │  │
│  │ └─────────────────────────────────────────────────────────────────┘  │  │
│  └───────────────────────────────────────────────────────────────────────┘  │
│                              ↓                                              │
│  ┌───────────────────────────────────────────────────────────────────────┐  │
│  │ Database Query Execution                                             │  │
│  │ ┌─────────────────────────────────────────────────────────────────┐  │  │
│  │ │ SELECT p.*, c.name, w.name FROM products p                     │  │  │
│  │ │ LEFT JOIN categories c ON p.category_id = c.id                 │  │  │
│  │ │ LEFT JOIN warehouses w ON p.warehouse_id = w.id                │  │  │
│  │ │ WHERE p.deleted_at IS NULL                      [SoftDelete]   │  │  │
│  │ │   AND (p.name LIKE '%laptop%' OR p.code LIKE '%laptop%')       │  │  │
│  │ │   AND p.category_id = 1                         [Filter]       │  │  │
│  │ │   AND p.warehouse_id = 1                        [Filter]       │  │  │
│  │ │ ORDER BY p.created_at DESC                      [Sort]         │  │  │
│  │ │ LIMIT 50 OFFSET 0                               [Pagination]   │  │  │
│  │ └─────────────────────────────────────────────────────────────────┘  │  │
│  └───────────────────────────────────────────────────────────────────────┘  │
│                              ↓                                              │
│  ┌───────────────────────────────────────────────────────────────────────┐  │
│  │ ProductResource (app/Http/Resources/ProductResource.php)             │  │
│  │ ┌─────────────────────────────────────────────────────────────────┐  │  │
│  │ │ Transforms each Product model:                                 │  │  │
│  │ │   - Convert snake_case to camelCase                            │  │  │
│  │ │   - Include relationships (category, warehouse names)          │  │  │
│  │ │   - Type cast values (int, float, string)                      │  │  │
│  │ │   - Format timestamps                                          │  │  │
│  │ │                                                                  │  │  │
│  │ │ Input (Model):                 Output (Resource):              │  │  │
│  │ │ {                               {                              │  │  │
│  │ │   id: 1,                         "id": "1",                    │  │  │
│  │ │   code: "PROD-001",       →      "code": "PROD-001",           │  │  │
│  │ │   buy_price: 100000,             "buyPrice": 100000,           │  │  │
│  │ │   category_id: 1,                "categoryId": "1",            │  │  │
│  │ │   warehouse_id: 1                "warehouseId": "1"            │  │  │
│  │ │ }                                }                              │  │  │
│  │ └─────────────────────────────────────────────────────────────────┘  │  │
│  └───────────────────────────────────────────────────────────────────────┘  │
│                              ↓                                              │
│  ┌───────────────────────────────────────────────────────────────────────┐  │
│  │ Response JSON                                                        │  │
│  │ ┌─────────────────────────────────────────────────────────────────┐  │  │
│  │ │ {                                                               │  │  │
│  │ │   "data": [                                                     │  │  │
│  │ │     {                                                           │  │  │
│  │ │       "id": "1",                                                │  │  │
│  │ │       "code": "PROD-001",                                       │  │  │
│  │ │       "name": "Laptop Dell XPS",                                │  │  │
│  │ │       "category": "Electronics",                                │  │  │
│  │ │       "categoryId": "1",                                        │  │  │
│  │ │       "buyPrice": 100000,                                       │  │  │
